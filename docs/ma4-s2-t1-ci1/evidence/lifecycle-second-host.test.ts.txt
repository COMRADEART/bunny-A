import test from "node:test";
import assert from "node:assert/strict";
import { mkdirSync, mkdtempSync, rmSync } from "node:fs";
import { tmpdir } from "node:os";
import { join } from "node:path";
import { createServer } from "node:net";
import { startHost } from "./http.server.ts";
import type { Executor } from "./ma4/task-runtime.server.ts";
const sample = async () => ({ at: Date.now(), cpu: { utilization: null, clockMhz: null, temperatureC: null }, memory: { usedBytes: 0, availableBytes: 1, totalBytes: 1 }, gpus: [], notes: [] });
const runs: string[] = []; let release: (() => void) | null = null;
const slow: Executor = { kind: "local.read", category: "read_only", gpu: false, sideEffect: false, timeoutMs: 30_000,
  async run(input, ctx) { runs.push(`${ctx.instanceId ?? "?"}:${ctx.stepId}`); if (input.wait) await new Promise<void>(r => { release = r; ctx.signal.addEventListener("abort", () => r(), { once: true }); }); return { ok: true, output: "x" }; } };
test("CI1 lifecycle: a second Host on the same data and port is refused and changes no running task", async () => {
  const s = createServer(); await new Promise<void>(r => s.listen(0, "127.0.0.1", r)); const port = (s.address() as { port: number }).port; await new Promise<void>(r => s.close(() => r()));
  const dir = mkdtempSync(join(tmpdir(), "ci1-same-")); mkdirSync(join(dir, "p"));
  const opts = { dataDirectory: dir, root: join(dir, "p"), port, adapters: [], sample, ptasks: { executors: [slow], call: async () => ({ ok: false as const, error: "no", durationMs: 0 }), modelAvailable: async () => false } };
  const first = await startHost(opts);
  const h = { connection: "close", Authorization: `Bearer ${first.token}`, "Content-Type": "application/json" };
  const state = async () => (await (await fetch(`http://127.0.0.1:${port}/state`, { headers: h })).json() as any).persistentTasks.tasks[0];
  try {
    await state();
    const env = { capabilities: ["local.read"], providers: ["ollama"], localOnly: true, models: [], projectScopes: ["p1"], sideEffects: false, background: false, budgets: { maxModelCalls: 0, maxWallMs: 3_600_000 }, expiresAt: Date.now() + 86_400_000 };
    const created = await (await fetch(`http://127.0.0.1:${port}/command`, { method: "POST", headers: h, body: JSON.stringify({ action: "ptask.create", data: { spec: { title: "Held", projectId: "p1", steps: [{ id: "w", kind: "local.read", input: { wait: true } }, { id: "z", kind: "local.read", input: {} }], acceptance: { requiredSteps: ["w", "z"] } }, envelope: env } }) })).json() as any;
    for (let i = 0; i < 100 && !release; i++) await new Promise(r => setTimeout(r, 20));
    const before = await state(); console.log("before second launch:", before.state, "version", before.version ?? "-");
    await assert.rejects(startHost(opts), (e: any) => { console.log("second Host refused:", e.code ?? e.message); return true; });
    const during = await state(); console.log("after refused launch:", during.state, "version", during.version ?? "-");
    assert.equal(during.state, before.state); assert.deepEqual({ ...during, at: 0 }, { ...before, at: 0 });
    release!();
    for (let i = 0; i < 100 && (await state()).state !== "COMPLETED"; i++) await new Promise(r => setTimeout(r, 20));
    assert.equal((await state()).state, "COMPLETED"); console.log("runs:", JSON.stringify(runs));
    assert.deepEqual(runs.map(r => r.split(":")[1]), ["w", "z"], "each step ran once, on the first Host");
  } finally { await first.close(); try { rmSync(dir, { recursive: true, force: true, maxRetries: 10 }); } catch (e: any) { console.log("cleanup:", e.code); } }
});
